// Injection seam for offline protocol tests; the running server uses its own service.
export async function handle(m,{signal,service=defaultService,cancelLocalTurn}){
  try{
    throwIfAdvisorCancelled(signal);
    let result;
    if(m.method==='initialize')result={protocolVersion:m.params?.protocolVersion??'2025-06-18',capabilities:{tools:{}},serverInfo:{name:'advisor-local',version:ADVISOR_RUNTIME_VERSION},instructions:'For suitable ordinary engineering work consider Halo Scout/reviewer, R9700 builder/tester, and the independently owned on-demand RTX3090 worker before paid auxiliaries. RTX3090 Scout/reviewer is read-only; builder/tester requires qualification in both its profile manifest and registry. Its large and small profiles share one GPU reservation with EVIE and Nemotron, so they cannot run concurrently. Prefer execution inference with exact supplied sourceFiles when the assignment is self-contained and needs no file discovery, edits, commands, browser, docs, web, or retrieval. Use execution tools only when one of those capabilities is materially required, with exact sourceFiles and allowedFiles, parent-selected test commands, and relevant toolServices. Calls on independent hardware may run concurrently. The host owns token budgets, output allowances, model-profile selection and memory checks. Omit maxOutputTokens and profile in ordinary delegation. Send concise goals, exact owned files and checks; workers can inspect implementation details. Do not estimate code lines, token counts, VRAM or fallback sizes. Dispatch after the first targeted inspection; respond to an actual host rejection instead of speculating about one. Use delegate_parallel_workers for disjoint R9700 and RTX3090 assignments. Explicit halo-only Scout/reviewer long-prefill workflows may use up to 5400000 ms; R9700, RTX3090 and other routes remain capped at 1200000 ms. Ordinary defaults and individual tool and command deadlines remain separate. Use local_status without starting a model. Primary model/effort stays selected; source-bound results are advisory. Inspect isolated changes, check original hashes, integrate and verify in primary. On CLOUD_CONTINUATION_REQUIRED continue via native Codex, never an implicit paid API. Honor local-only/cloud-only, exact-worker-only modes, and explicit RTX3090 profile choices. local_status reports each resource.'};
    else if(m.method==='ping')result={};
    else if(m.method==='tools/list')result={tools:toolDefinitions.filter(t=>!gaiaProfile||gaiaTools.has(t.name))};
    else if(m.method==='tools/call'){
      const name=m.params?.name;let value;
      if(gaiaProfile&&!gaiaTools.has(name))throw new Error('UNKNOWN_TOOL');
      if(name==='cancel_local_turn'){
        if(typeof cancelLocalTurn!=='function')throw Object.assign(new Error('CANCELLATION_CONTEXT_REQUIRED'),{code:'CANCELLATION_CONTEXT_REQUIRED'});
        value=await cancelLocalTurn(m.params.arguments);
        // Interrupt hooks accept their lifecycle output schema only. Keep the
        // diagnostic counts structured; empty JSON text is a valid hook result.
        return {jsonrpc:'2.0',id:m.id,result:{content:[{type:'text',text:'{}'}],structuredContent:value}};
      }
      if(name==='delegate_parallel_workers'&&Array.isArray(m.params.arguments?.requests))m.params.arguments={...m.params.arguments,requests:m.params.arguments.requests.map(inspections.constrainRequest)};
      else if(name?.startsWith('delegate_')||name==='preflight_local_request')m.params.arguments=inspections.constrainRequest(name==='delegate_project_to_halo'?{...m.params.arguments,execution:'tools'}:m.params.arguments);
      if(gaiaProfile&&['delegate_to_r9700','delegate_to_r9700b','delegate_to_rtx3090','preflight_local_request'].includes(name)){
        const request=m.params.arguments??{},worker=name==='preflight_local_request'?request.role?.split('_')[0]:name.slice('delegate_to_'.length);
        if(!['r9700','r9700b','rtx3090'].includes(worker)||request.mode!==`${worker}-only`||!request.role?.startsWith(`${worker}_`)||(request.engine!==undefined&&request.engine!=='direct'))throw Object.assign(new Error('EXPLICIT_WORKER_ONLY_REQUIRED'),{code:'EXPLICIT_WORKER_ONLY_REQUIRED'});
      }
      if(['context7_docs','cerebras_research'].includes(name)){
        const args=m.params.arguments??{};
        const keys=name==='context7_docs'?['directory','libraryName','query']:['directory','task'];
        if(Object.keys(args).length!==keys.length||Object.keys(args).some(k=>!keys.includes(k))||typeof args.directory!=='string')throw new Error('INVALID_LOOKUP_REQUEST');
        const {directory,...question}=args;
        const lookup=await createAdvisorToolServices({signal,workspaceRoot:directory,privacyState:inspections.privacyFor(directory),options:{docs:true,research:true,asNeeded:true}});
        try{value=await lookup.execute(name==='context7_docs'?'docs_query':name,question);}finally{await lookup.close();}
      }
      else if(name==='inspection_discover')value=await inspections.discover(m.params.arguments);
      else if(name==='inspection_call')value=await inspections.call(m.params.arguments);
      else if(name==='local_status')value=await service.status({signal});
      else if(name==='apply_worker_result')value=await applyAdvisorResult(m.params.arguments??{},service);
      else if(name==='delegate_project_to_halo'){
        const {engine='direct',...request}=m.params.arguments??{};
        if(engine!=='direct')throw new Error('COORDINATOR_ENGINE_INVALID');
        value=await service.run({timeoutMs:1200000,maxOutputTokens:8192,maxSteps:24,maxToolCalls:48,...request,role:'halo_coordinator',mode:'halo-only',worker:'halo',execution:'tools'},{signal});
      }
      else if(name==='delegate_parallel_workers'){
        const args=m.params.arguments;
        if(!args||typeof args!=='object'||Array.isArray(args)||Object.keys(args).length!==1||!Object.hasOwn(args,'requests'))throw Object.assign(new Error('INVALID_PARALLEL_REQUEST'),{code:'INVALID_PARALLEL_REQUEST'});
        value=await delegateParallelWorkers({service,requests:args.requests,signal});
      }
      else {
        const worker={delegate_to_halo:'halo',delegate_to_r9700:'r9700',delegate_to_r9700b:'r9700b',delegate_to_rtx3090:'rtx3090',delegate_to_local_fleet:'fleet'}[name];if(!worker&&name!=='preflight_local_request')throw new Error('UNKNOWN_TOOL');
        const {engine='direct',...request}=m.params.arguments??{};
        if(!['direct','hermes'].includes(engine))throw new Error('UNKNOWN_ENGINE');
        let selected=service;
        if(engine==='hermes'){
          if(request.execution!=='tools')throw Object.assign(new Error('HERMES_REQUIRES_TOOLS'),{code:'HERMES_REQUIRES_TOOLS'});
          const {runHermesToolLoop}=await import('file:////SETUP_REQUIRED/path-1');
          selected=new AdvisorLocal({toolLoop:runHermesToolLoop});
        }
        value=name==='preflight_local_request'?await selected.preflight(request,{signal}):await selected.run({...request,worker},{signal,privacyState:sharedPrivacy});
      }
      result={content:[{type:'text',text:JSON.stringify(value)}],structuredContent:value,isError:['ERROR','REQUEST_REPAIR_REQUIRED','PREFLIGHT_FAILED','FAILED','CANCELLED'].includes(value.status)};
    }else{return {jsonrpc:'2.0',id:m.id,error:{code:-32601,message:'Method not found'}};}
    return {jsonrpc:'2.0',id:m.id,result};
  }catch(e){const value=advisorRequestError(e);return {jsonrpc:'2.0',id:m.id,result:{content:[{type:'text',text:JSON.stringify(value)}],structuredContent:value,isError:true}};}
}
const requests=createMcpRequestDispatcher({execute:handle,send});
const close=async()=>{await requests.close();await inspections.close();};
const entryPath=process.argv[1]?path.resolve(process.argv[1]):'';
const modulePath=fileURLToPath(import.meta.url);
if((process.platform==='win32'?entryPath.toLowerCase()===modulePath.toLowerCase():entryPath===modulePath)){
process.stdin.on('end',()=>{void close();});
process.stdin.on('error',()=>{void close();});
// Each request has its own promise; never await a model inside the framing loop.
let buffer='';process.stdin.setEncoding('utf8');process.stdin.on('data',chunk=>{
  buffer+=chunk;if(Buffer.byteLength(buffer)>1048576){buffer='';return;}
  let index;while((index=buffer.indexOf('\n'))>=0){const line=buffer.slice(0,index);buffer=buffer.slice(index+1);try{void requests.dispatch(JSON.parse(line));}catch{send({jsonrpc:'2.0',id:null,error:{code:-32700,message:'Parse error'}});}}
});
